cmake_minimum_required(VERSION 3.28)
project(crux_messaging_consumer LANGUAGES C)
if(NOT CRUX_SOURCE_DIR)
    get_filename_component(CRUX_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CRUX_BUILD_THREAD ON CACHE BOOL "Required external capability")
set(CRUX_BUILD_TIME ON CACHE BOOL "Thread's clock dependency")
set(CRUX_TIME_PROVIDER memory CACHE STRING "Controlled development realtime")
add_subdirectory("${CRUX_SOURCE_DIR}" crux)
include("${CRUX_SOURCE_DIR}/cmake/CruxConsumerBoundary.cmake")
crux_assert_portable_usage(crux::time crux::thread)
add_library(messaging_controller_portable OBJECT scenario/controller.c)
target_link_libraries(messaging_controller_portable PRIVATE crux::time crux::thread)
# Only the composition selects native providers; no provider usage is exposed to
# the independently compiled C17 portable source target above.
add_executable(messaging_controller $<TARGET_OBJECTS:messaging_controller_portable>)
target_link_libraries(messaging_controller PRIVATE crux::time crux::thread)
target_compile_options(messaging_controller_portable PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(messaging_controller_portable PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES
    C_EXTENSIONS OFF)
crux_assert_portable_consumer(messaging_controller_portable)
enable_testing()
add_test(NAME Messaging.ControllerConsumer COMMAND messaging_controller)
set_tests_properties(Messaging.ControllerConsumer PROPERTIES TIMEOUT 20)
